<script>

let records = JSON.parse(localStorage.getItem('fuel')) || [];

function save(){localStorage.setItem('fuel', JSON.stringify(records))}

function add(){

    const mile = Number(document.getElementById('mile').value);

    const litre = Number(document.getElementById('litre').value);

    const price = Number(document.getElementById('price').value);

    const note = document.getElementById('note').value.trim();

    if(!mile || !litre || !price){alert('請填齊里程、油量、油價');return;}

    const last = records[records.length-1];

    let dist = null, fuel100 = null, costKm = null;

    if(last){

        dist = mile - last.mile;

        fuel100 = (litre / dist *100).toFixed(2);

        costKm = (litre * price / dist).toFixed(3);

    }

    records.push({

        date: new Date().toLocaleString('zh-Hant'),

        mile,litre,price,dist,fuel100,costKm,note

    });

    save();render();

    document.getElementById('mile').value='';

    document.getElementById('litre').value='';

    document.getElementById('price').value='';

    document.getElementById('note').value='';

    alert('新增成功！');

}

function render(){

    const box = document.getElementById('list');

    const statBox = document.getElementById('stat');

    box.innerHTML = '';

    if(records.length===0){box.innerHTML='暫時冇記錄';statBox.innerText='平均油耗：- L/100km｜每公里油費：- MOP';return;}

    let sumFuel=0,sumDist=0;

    records.forEach((r,i)=>{

        if(r.fuel100){sumFuel += Number(r.fuel100);sumDist += Number(r.dist);}

        box.innerHTML += `<div class="item">${r.date}<br>里程：${r.mile}km｜加${r.litre}L@${r.price}MOP/L ${r.dist?'｜行${r.dist}km，油耗${r.fuel100} L/100km，每公里${r.costKm} MOP':''}${r.note?'<br>備註：'+r.note:''} <button onclick="del(${i})">刪除</button></div>`;

    })

    const avgFuel = sumDist>0 ? (sumFuel / records.filter(x=>x.fuel100).length).toFixed(2) : '-';

    const avgCost = sumDist>0 ? (sumFuel * records[records.length-1].price / sumDist).toFixed(3) : '-';

    statBox.innerText = `平均油耗：${avgFuel} L/100km｜每公里油費：${avgCost} MOP`;

}

function del(i){

    if(!confirm('刪除呢條？'))return;

    records.splice(i,1);save();render();

}

function clearAll(){

    if(!confirm('確定清空所有記錄？冇得返轉頭！'))return;

    records=[];save();render();

}

function exportCSV(){

    if(records.length===0){alert('暫時冇記錄');return;}

    let csv = "日期,總里程(km),加油量(L),油價(MOP/L),行車距離(km),油耗(L/100km),每公里(MOP),備註\n";

    records.forEach(r=>csv += `${r.date},${r.mile},${r.litre},${r.price},${r.dist||''},${r.fuel100||''},${r.costKm||''},${r.note||''}\n`);

    const blob = new Blob([csv], {type:'text/csv;charset=utf-8'});

    const url = URL.createObjectURL(blob);

    const a = document.createElement('a');

    a.href=url;a.download="AFR125X油耗記錄.csv";a.click();

    URL.revokeObjectURL(url);

}

render();
